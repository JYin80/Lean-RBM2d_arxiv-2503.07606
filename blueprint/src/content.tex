\chapter{The model and the propagator}

\begin{definition}[Block covariance matrix, \S2.1]\label{def:SB}
  \lean{RBM.SB}\leanok
  Let $L\ge 3$ and work on the discrete torus $\ZL$.  Set
  \[ \SB_{ab} \;=\; \tfrac15\,\mathbf 1\!\left(|a-b|_L\le 1\right),\qquad a,b\in\ZL, \]
  the transition matrix of the lazy nearest-neighbour random walk on $\ZL$.
  Equivalently, $\SB$ is the circulant matrix on the group $\ZL$ generated by the
  kernel supported on $\{(0,0),(\pm1,0),(0,\pm1)\}$ with value $1/5$.
\end{definition}

\begin{remark}[Why $L\ge3$]\label{rem:L3}
  The paper leaves the hypothesis $L\ge 3$ implicit.  It is needed: for $L\le 2$
  the five lattice points above are not distinct and $\SB$ fails to be stochastic.
  See \texttt{docs/paper-deltas.md}, item 1.
\end{remark}

\begin{definition}[Periodic distance]\label{def:dist}
  \lean{RBM.zdist2}\leanok
  $|x|_L$ denotes the periodic distance on $\ZL$.  The paper uses the periodic
  Euclidean norm and notes that the periodic $L^1$ and $L^2$ norms differ by at
  most $\sqrt2$, which is absorbed into the constants; we formalize the $L^1$
  one.  See \texttt{docs/paper-deltas.md}, item 2.
\end{definition}

\begin{lemma}[Elementary properties of $\SB$]\label{lem:SB-basic}
  \lean{RBM.SB_isSymm, RBM.SB_apply_add_right, RBM.sum_SB_row, RBM.SB_mulVec_one,
    RBM.card_sbSupport}\leanok
  \uses{def:SB}
  $\SB$ is symmetric and translation invariant, each row sums to $1$, and
  $\SB\mathbf 1=\mathbf 1$.
\end{lemma}
\begin{proof}\leanok
  Symmetry and translation invariance are immediate from the circulant form.  The
  row sums reduce, after the change of variables $b\mapsto a-b$, to the total mass
  of the kernel, which is $5\cdot\frac15=1$ because the five lattice points
  $(0,0),(\pm1,0),(0,\pm1)$ are distinct when $L\ge3$.
\end{proof}

\begin{definition}[The propagator, Definition \texttt{def\_Theta}]\label{def:Theta}
  \lean{RBM.Theta}\leanok\uses{def:SB}
  For $\xi\in\CC$ with $|\xi|<1$,
  \[ \Th^{(B)}_\xi := \frac{1}{1-\xi\,\SB}. \]
  In Lean this is \texttt{Ring.inverse} of $1-\xi\SB$ in the matrix ring with the
  $\ell^\infty$ operator norm; $\|\SB\|=1$ makes $1-\xi\SB$ a unit.
\end{definition}

\begin{lemma}[$\Th$ is the unique inverse]\label{lem:Theta-unique}
  \lean{RBM.Theta_mul, RBM.mul_Theta, RBM.eq_Theta_of_mul}\leanok\uses{def:Theta}
  $\Th^{(B)}_\xi$ is a two-sided inverse of $1-\xi\SB$, and any left inverse of
  $1-\xi\SB$ equals it.  This is the workhorse behind properties 1--3.
\end{lemma}
\begin{proof}\leanok
  $\SB$ is stochastic and symmetric, so $\|\SB\|=1$ in the $\ell^\infty$ operator
  norm; with $|\xi|<1$ this makes $1-\xi\SB$ a unit.  Uniqueness is the one-line
  computation $B=B\,(1-\xi\SB)\,\Th_\xi=\Th_\xi$.
\end{proof}

\begin{lemma}[Properties 1--4 of Lemma \texttt{lem\_propTH}]\label{lem:propTH-1234}
  \lean{RBM.Theta_transpose, RBM.Theta_apply_add_right, RBM.Theta_commute_SB,
    RBM.Theta_commute, RBM.Theta_eq_tsum, RBM.sum_Theta_row}\leanok
  \uses{def:Theta, lem:Theta-unique, lem:SB-basic}
  $\Th^{(B)}_\xi$ is symmetric and translation invariant, commutes with $\SB$ and
  with $\Th^{(B)}_{\xi'}$, and for $|\xi|<1$ admits the random-walk representation
  $\Th^{(B)}_\xi=\sum_{k\ge0}\xi^k(\SB)^k$.  Moreover
  $\sum_b(\Th^{(B)}_\xi)_{ab}=(1-\xi)^{-1}$.
\end{lemma}
\begin{proof}\leanok
  Properties 1--3 each transfer from the corresponding property of $\SB$ by
  uniqueness of the inverse: transpose, conjugate by the translation
  $a\mapsto a+s$, and commute.  Property 4 is the Neumann series, convergent in
  operator norm because $\|\xi\SB\|<1$.  The row sum follows from
  $\SB\mathbf1=\mathbf1$.
\end{proof}

\begin{lemma}[Crude bounds]\label{lem:crude}
  \lean{RBM.norm_Theta_le, RBM.sum_norm_Theta_row_le, RBM.norm_Theta_apply_le}\leanok
  \uses{lem:propTH-1234}
  $\max_a\sum_b|(\Th^{(B)}_\xi)_{ab}|\le(1-|\xi|)^{-1}$.  This is \emph{not} the
  sharp bound: the decay length here is governed by $1-|\xi|$, whereas the paper
  needs $\lhat(\xi)=\min(|1-\xi|^{-1/2},L)$.  The two differ enormously when
  $\xi=tm^2$.
\end{lemma}
\begin{proof}\leanok
  Bound the Neumann series term by term in the $\ell^\infty$ operator norm.
\end{proof}

\begin{definition}[Deterministic domination, Definition \texttt{stoch\_domination} (ii)]
  \label{def:detdom}
  \lean{RBM.UnifDetDom, RBM.DetDom}\leanok
  $A\prec B$ means that for every fixed $\varepsilon>0$ one has
  $A\le C_\varepsilon L^{\varepsilon}B$ for all large $L$; in particular powers of
  $\log L$ are harmless.  The sequence parameter is the number of blocks $L$,
  which is the form \S8 uses; the introduction's $N^{\tau}$ version follows since
  $L\le N$.  The basic notion is the version uniform in a parameter
  $u\in U(L)$, because Properties 5 and 6 hold uniformly in the lattice points.
  See \texttt{docs/paper-deltas.md}, items 3 and 4.
\end{definition}


\begin{definition}[Scales, \texttt{(eq\_kappa\_def)}]\label{def:kappa}
  \lean{RBM.kappa, RBM.ellhat}\leanok
  $\kappa:=|1-\xi|^{1/2}$ and $\lhat=\lhat(\xi):=\min(\kappa^{-1},L)$.
\end{definition}

\begin{lemma}[Elementary properties of $\kappa$ and $\lhat$]\label{lem:kappa-basic}
  \lean{RBM.kappa_sq, RBM.kappa_pos, RBM.kappa_mul_ellhat_le_one}\leanok\uses{def:kappa}
  $\kappa^2=|1-\xi|$, $\kappa>0$ for $|\xi|<1$, and $\kappa\lhat\le1$.
\end{lemma}
\begin{proof}\leanok
  $\kappa>0$ because $|\xi|<1$ forces $\xi\ne1$, and
  $\kappa\lhat\le\kappa\cdot\kappa^{-1}=1$.
\end{proof}

\section{Fourier setup}

\begin{definition}[The symbol, \texttt{(eq\_symbol)}]\label{def:Shat}
  \lean{RBM.Shat}\leanok\uses{def:SB}
  For $p\in\TL$,
  $\Shat(p)=\frac{1+2\cos p_1+2\cos p_2}{5}$.
\end{definition}

\begin{lemma}[Plane waves diagonalize $\SB$]\label{lem:eigen}
  \lean{RBM.Shat_eq_cos, RBM.SB_mulVec_char, RBM.norm_Shat_le_one,
    RBM.one_sub_mul_Shat_ne_zero}\leanok
  \uses{def:Shat, def:SB}
  $\SB e_p=\Shat(p)e_p$, $|\Shat(p)|\le1$, and $1-\xi\Shat(p)\ne0$ for $|\xi|<1$.
\end{lemma}
\begin{proof}\leanok
  Expanding $\SB$ over its five-point support turns each shift into a character
  value, which is exactly $\Shat(p)$.  Then $|\Shat(p)|\le1$ gives
  $|\xi\Shat(p)|<1$, so $1-\xi\Shat(p)\ne0$.
\end{proof}

\begin{lemma}[Fourier representation, \texttt{(eq\_Fourier\_rep)}]\label{lem:fourier}
  \lean{RBM.Theta_apply_fourier, RBM.inv_mul_sum_chr}\leanok
  \uses{lem:eigen, lem:Theta-unique}
  With $x=a-b$,
  \[ (\Th^{(B)}_\xi)_{ab}=K_{\xi,L}(x)=\frac1{L^2}\sum_{p\in\TL}
     \frac{e^{\mathrm{i}p\cdot x}}{1-\xi\Shat(p)} . \]
\end{lemma}
\begin{proof}\leanok
  Character orthogonality on $\ZL$ shows that the kernel $K_{\xi,L}$ solves
  $K-\xi\SB K=\delta_0$; since $\Th_\xi$ is the unique inverse of $1-\xi\SB$, the
  circulant matrix generated by $K_{\xi,L}$ is $\Th_\xi$.
\end{proof}

\begin{definition}[The symbol gap $q(p)$, \texttt{(eq\_qdef)}]\label{def:q}
  \lean{RBM.qsym}\leanok\uses{def:Shat}
  $q(p):=1-\Shat(p)=\frac25\big[(1-\cos p_1)+(1-\cos p_2)\big]$.
\end{definition}

\begin{lemma}[Ellipticity, \texttt{(eq\_elliptic)}]\label{lem:elliptic}
  \lean{RBM.norm_one_sub_mul_Shat_le, RBM.norm_one_sub_mul_Shat_ge,
    RBM.norm_one_sub_mul_Shat_ge_kappa}\leanok
  \uses{def:q, def:kappa}
  For $|\xi|<1$,
  \[ \tfrac19\big(\kappa^2+q(p)\big)\;\le\;\big|1-\xi\Shat(p)\big|\;\le\;\kappa^2+q(p). \]
  The paper's proof splits at $\Shat(p)=1/2$; in the low branch its constant
  $1/2$ should read $2/5$, see \texttt{docs/paper-deltas.md}, item 5.
\end{lemma}
\begin{proof}\leanok
  Write $\lambda=\Shat(p)$, which is real and lies in $[-3/5,1]$.  If
  $\lambda\le1/2$ the multiplier is bounded away from $1$ outright.  If
  $\lambda\ge1/2$, use $1-\lambda\xi=(1-\lambda)+\lambda(1-\xi)$: since
  $|\xi|<1$ we have $\operatorname{Re}(1-\xi)>0$, so the real parts of the two
  summands cannot cancel and $|1-\lambda\xi|$ dominates both
  $(1-\lambda)+\lambda\operatorname{Re}(1-\xi)$ and
  $\lambda|\operatorname{Im}(1-\xi)|$.  The upper bound is the triangle
  inequality and needs no hypothesis on $\xi$ at all.
\end{proof}

\begin{definition}[$|p|_*$ on the lattice]\label{def:pstar}
  \uses{def:dist}
  \lean{RBM.pstar, RBM.pstar2}\leanok
  On $\TL$ the frequency in coordinate $i$ is $2\pi p_i/L$, and there
  $\operatorname{dist}(2\pi p_i/L,2\pi\mathbb Z)=2\pi\,\mathrm{zdist}(p_i)/L$
  \emph{exactly}.  So $|p|_*$ is defined straight from the cycle distance and
  $\operatorname{dist}(\cdot,2\pi\mathbb Z)$ is never formalized; the single
  identity carrying the periodicity is \texttt{RBM.cos\_eq\_cos\_pstar}.
\end{definition}

\begin{lemma}[$q(p)\sim|p|_*^2$, \texttt{(eq\_qcomp)}]\label{lem:qcomp}
  \uses{def:q, def:pstar}
  \lean{RBM.qsym_le_pstar2, RBM.pstar2_le_qsym,
        RBM.norm_one_sub_mul_Shat_le_pstar, RBM.norm_one_sub_mul_Shat_ge_pstar}\leanok
  $q(p)\sim|p|_*^2$, where $|p|_*^2=\operatorname{dist}(p_1,2\pi\mathbb Z)^2
  +\operatorname{dist}(p_2,2\pi\mathbb Z)^2$; equivalently
  $|1-\xi\widehat S(p)|\sim\kappa^2+|p|_*^2$, which is the form \S8.2 and \S8.3 use.
  Explicitly $\tfrac{4}{5\pi^2}|p|_*^2\le q(p)\le\tfrac15|p|_*^2$.
\end{lemma}
\begin{proof}\leanok
  By \texttt{RBM.cos\_eq\_cos\_pstar} the cosine does not see the difference
  between $2\pi p_i/L$ and its distance to $2\pi\mathbb Z$, so each coordinate
  reduces to the elementary two-sided bound
  $\tfrac{2}{\pi^2}\theta^2\le1-\cos\theta\le\tfrac12\theta^2$ on $|\theta|\le\pi$,
  the hypothesis being supplied by $\mathrm{zdist}\le L/2$.  Merging with
  \eqref{lem:elliptic} costs only the numeric fact $4/(45\pi^2)\le1/9$, i.e.\ $\pi>3$.
\end{proof}

\begin{lemma}[Shell counting on $\TL$]\label{lem:shells}
  \uses{def:dist}
  With $k(p)=\max(\mathrm{zdist}(p_1),\mathrm{zdist}(p_2))$ one has
  $\#\{p\in\TL: k(p)=k\}\le12k+4$, and every sum over $\TL\setminus\{0\}$
  splits into the shells $k=1,\dots,L$.  Pure \texttt{Finset} accounting: no reals.
  \emph{Work order T15.}
\end{lemma}

\begin{lemma}[Harmonic sum]\label{lem:harmonic}
  \uses{def:detdom}
  \lean{RBM.mul_log_le_rpow, RBM.sum_inv_Icc_le_one_add_log,
        RBM.one_add_log_detDom_one, RBM.harmonic_detDom_one}
  $\sum_{k=1}^{L}k^{-1}\le1+\log L\prec1$, the Lean form of the paper's remark
  that powers of $\log L$ are harmless.  Mathlib supplies the first inequality
  over $\mathbb Q$ (\texttt{harmonic\_le\_one\_add\_log}); the domination is
  $\tau\log x\le x^{\tau}$, itself $\log y\le y-1$ at $y=x^{\tau}$.
  \emph{Work order T16 --- drafted, awaiting a CI round; no
  \texttt{\textbackslash leanok} until it compiles.}
\end{lemma}

\begin{lemma}[Lattice sum]\label{lem:latticesum}
  \uses{lem:qcomp, lem:shells, lem:harmonic}
  $\sum_{p\in\TL\setminus\{0\}}|p|_*^{-2}\le CL^2\log L$.
  \emph{Work order T3.}
\end{lemma}

\begin{lemma}[Separating the zero mode]\label{lem:zero-mode}
  \uses{lem:fourier}
  In \texttt{(eq\_Fourier\_rep)} the $p=0$ term is $L^{-2}(1-\xi)^{-1}$, and in a
  difference $K_{\xi,L}(x)-K_{\xi,L}(y)$ it cancels outright.  Both Property 5
  and Property 6 begin with exactly this step, so it is proved once.
  \emph{Work order T14.}
\end{lemma}

\section{Property 5: exponential decay}

\begin{lemma}[Decay in the regime $\kappa L<1$]\label{lem:decay-small}
  \uses{lem:fourier, lem:elliptic, lem:latticesum, lem:zero-mode}
  If $\kappa L<1$, so that $\lhat=L$, then separating the zero mode gives
  $|K_{\xi,L}(x)|\le\kappa^{-2}L^{-2}+C\log L\prec\kappa^{-2}L^{-2}$, which is
  \texttt{(prop:ThfadC)} because $|x|_L\le CL=C\lhat$.
  \emph{Work order T4.}
\end{lemma}

\begin{lemma}[Infinite-volume kernel and the contour shift]\label{lem:contour}
  \uses{lem:elliptic}
  $|D_\xi(p+\mathrm{i}\eta)|\ge c(\kappa^2+|p|_*^2)$ for $|\eta|\le c_0\kappa$, and
  hence $|K_{\xi,\infty}(x)|\le C\log(2+\kappa^{-1})e^{-c\kappa|x|}$.
  \emph{Work order T9.}
\end{lemma}

\begin{lemma}[Periodization]\label{lem:periodize}
  \uses{lem:contour, lem:fourier}
  $K_{\xi,L}(x)=\sum_{n\in\mathbb Z^2}K_{\xi,\infty}(x+nL)$, by comparing Fourier
  coefficients rather than by invoking Poisson summation.
  \emph{Work order T10.}
\end{lemma}

\begin{lemma}[Property 5, \texttt{(prop:ThfadC)}]\label{lem:propTH-5}
  \uses{lem:decay-small, lem:periodize}
  $(\Th^{(B)}_\xi)_{ab}\prec e^{-c|a-b|_L/\lhat(\xi)}\big/\big(|1-\xi|\lhat(\xi)^2\big)$.
\end{lemma}

\chapter{Delocalization}

\begin{lemma}[Deterministic core of Theorem \texttt{MR:decol}]\label{lem:deloc}
  \lean{RBM.green, RBM.green_eq_spectral, RBM.im_green_apply_self,
    RBM.sq_norm_eigenvector_le_im_green, RBM.sq_norm_eigenvector_le_of_norm_green_le}\leanok
  For Hermitian $H$ with eigenpairs $(\lambda_k,\psi_k)$ and $\eta>0$,
  \[ |\psi_k(x)|^2\;\le\;\sum_l\frac{\eta^2|\psi_l(x)|^2}{(\lambda_k-\lambda_l)^2+\eta^2}
     \;=\;\eta\,\operatorname{Im}G_{xx}(\lambda_k+\mathrm{i}\eta), \]
  so a bound $|G_{xx}|\le C$ gives $|\psi_k(x)|^2\le C\eta$.  The probabilistic
  input (the local law \texttt{MR:locSC}) is a hypothesis here.
\end{lemma}
\begin{proof}\leanok
  In the middle sum the $l=k$ term is $|\psi_k(x)|^2$ and every other term is
  non-negative, which gives the inequality; the spectral decomposition of
  $G(\lambda_k+\mathrm{i}\eta)$ identifies the sum with
  $\eta\operatorname{Im}G_{xx}$.
\end{proof}

\begin{remark}[The random layer]\label{rem:random}
  Mathlib has no It\^o calculus for matrix-valued Brownian motion, no matrix SDEs
  and no Dyson Brownian motion.  The loop hierarchy of \S5--\S7 and the inductive
